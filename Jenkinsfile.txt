pipeline {
    agent any

    environment {
        S3_BUCKET = 's3-deployment-files'
        AWS_DEFAULT_REGION = 'ap-south-1'
        PIPELINE_NAME = 'ATHISH'
    }

    options {
        timestamps()
    }

    stages {

        stage('Checkout') {
            steps {

                script {
                    env.CURRENT_STAGE = "Checkout"
                    env.PIPELINE_START_TIME = new Date().format(
                        "yyyy-MM-dd'T'HH:mm:ss'Z'",
                        TimeZone.getTimeZone('UTC')
                    )
                }

                checkout scm

                script {
                    env.COMMIT_ID = sh(
                        script: "git rev-parse --short HEAD",
                        returnStdout: true
                    ).trim()

                    echo "Commit ID: ${env.COMMIT_ID}"
                    echo "Pipeline Start Time: ${env.PIPELINE_START_TIME}"
                }

            }
        }

        stage('Validate HTML') {
            steps {

                script {
                    env.CURRENT_STAGE = "HTML Validation"
                }

                sh '''
                echo "========================================"
                echo "Validating HTML..."
                echo "========================================"

                if [ ! -f index.html ]; then
                    echo "ERROR: index.html not found."
                    exit 1
                fi

                tidy -qe index.html

                echo "HTML Validation Passed."
                '''
            }
        }

        stage('Deploy Website') {
            steps {

                script {
                    env.CURRENT_STAGE = "Deploy"
                }

                sh '''
                echo "========================================"
                echo "Deploying Website to Amazon S3..."
                echo "========================================"

                aws s3 sync . s3://$S3_BUCKET \
                    --delete \
                    --exclude ".git/*" \
                    --exclude ".github/*" \
                    --exclude "*.zip" \
                    --exclude "Jenkinsfile" \
                    --exclude "Jenkinsfile.txt" \
                    --exclude "*.md" \
                    --exclude ".gitignore"

                echo "Deployment Completed Successfully."
                '''
            }
        }

        stage('Publish Event to EventBridge') {
            steps {
                script {

                    env.CURRENT_STAGE = "Publish Event"

                    def endTime = new Date()
                    def endTimeIso = endTime.format(
                        "yyyy-MM-dd'T'HH:mm:ss'Z'",
                        TimeZone.getTimeZone('UTC')
                    )

                    def duration = currentBuild.duration > 0 ?
                        (currentBuild.duration / 1000).toInteger() : 0

                    writeFile file: 'event.json', text: """
[
  {
    "Source": "pipewatch.jenkins",
    "DetailType": "Pipeline Status",
    "EventBusName": "default",
    "Detail": "{\\"pipelineName\\":\\"${env.PIPELINE_NAME}\\",\\"buildNumber\\":\\"${env.BUILD_NUMBER}\\",\\"branch\\":\\"main\\",\\"commitId\\":\\"${env.COMMIT_ID}\\",\\"stage\\":\\"${env.CURRENT_STAGE}\\",\\"status\\":\\"SUCCESS\\",\\"environment\\":\\"Production\\",\\"triggeredBy\\":\\"Jenkins\\",\\"startTime\\":\\"${env.PIPELINE_START_TIME}\\",\\"endTime\\":\\"${endTimeIso}\\",\\"duration\\":${duration}}"
  }
]
"""

                    sh '''
                    echo "Generated Event"
                    cat event.json
                    aws events put-events --entries file://event.json
                    '''

                }
            }
        }

    }

    post {

        success {
            echo "========================================"
            echo "Pipeline Status : SUCCESS"
            echo "Website Successfully Deployed"
            echo "========================================"
        }

        failure {

            script {

                def endTime = new Date()
                def endTimeIso = endTime.format(
                    "yyyy-MM-dd'T'HH:mm:ss'Z'",
                    TimeZone.getTimeZone('UTC')
                )
                def duration = currentBuild.duration > 0 ?
                    (currentBuild.duration / 1000).toInteger() : 0

                writeFile file: 'failure.json', text: """
[
  {
    "Source": "pipewatch.jenkins",
    "DetailType": "Pipeline Status",
    "EventBusName": "default",
    "Detail": "{\\"pipelineName\\":\\"${env.PIPELINE_NAME}\\",\\"buildNumber\\":\\"${env.BUILD_NUMBER}\\",\\"branch\\":\\"main\\",\\"commitId\\":\\"${env.COMMIT_ID ?: 'Unknown'}\\",\\"stage\\":\\"${env.CURRENT_STAGE}\\",\\"status\\":\\"FAILURE\\",\\"environment\\":\\"Production\\",\\"triggeredBy\\":\\"Jenkins\\",\\"startTime\\":\\"${env.PIPELINE_START_TIME ?: ''}\\",\\"endTime\\":\\"${endTimeIso}\\",\\"duration\\":${duration}}"
  }
]
"""

                sh '''
                aws events put-events --entries file://failure.json || true
                '''
            }

            echo "========================================"
            echo "Pipeline Status : FAILURE"
            echo "Deployment Cancelled"
            echo "========================================"
        }

        always {
            echo "========================================"
            echo "Pipeline Finished"
            echo "========================================"
            cleanWs()
        }
    }
}
