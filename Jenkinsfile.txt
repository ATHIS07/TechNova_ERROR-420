pipeline {
    agent any

    environment {
        S3_BUCKET = 's3-deployment-files'
        AWS_DEFAULT_REGION = 'ap-south-1'
    }

    options {
        timestamps()
    }

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Validate HTML') {
            steps {
                sh '''
                echo "========================================"
                echo "Validating HTML..."
                echo "========================================"

                if [ ! -f index.html ]; then
                    echo "ERROR: index.html not found."
                    exit 1
                fi

                tidy -qe index.html

                echo "HTML Validation Passed."
                '''
            }
        }

        stage('Deploy Website') {
            steps {
                sh '''
                echo "========================================"
                echo "Deploying Website to Amazon S3..."
                echo "========================================"

                aws s3 sync . s3://$S3_BUCKET \
                    --delete \
                    --exclude ".git/*" \
                    --exclude ".github/*" \
                    --exclude "*.zip" \
                    --exclude "Jenkinsfile" \
                    --exclude "Jenkinsfile.txt" \
                    --exclude "*.md" \
                    --exclude ".gitignore"

                echo "Deployment Completed Successfully."
                '''
            }
        }

    }

    post {

        success {

            echo "========================================"
            echo "Pipeline Status : SUCCESS"
            echo "========================================"

            sh """
            cat > event.json <<EOF
[
  {
    "Source":"pipewatch.jenkins",
    "DetailType":"Pipeline Status",
    "EventBusName":"default",
    "Detail":"{\\"PipelineName\\":\\"PipeWatch Frontend\\",\\"BuildNumber\\":\\"${BUILD_NUMBER}\\",\\"Branch\\":\\"main\\",\\"CurrentStage\\":\\"Deploy\\",\\"Status\\":\\"SUCCESS\\",\\"Environment\\":\\"Production\\",\\"TriggeredBy\\":\\"Jenkins\\",\\"StartTime\\":\\"$(date -u +"%Y-%m-%dT%H:%M:%SZ")\\",\\"Duration\\":0,\\"CommitID\\":\\"${GIT_COMMIT ?: 'Unknown'}\\",\\"JobName\\":\\"${JOB_NAME}\\",\\"BuildURL\\":\\"${BUILD_URL}\\"}"
  }
]
EOF

            aws events put-events --entries file://event.json
            """
        }

        failure {

            echo "========================================"
            echo "Pipeline Status : FAILED"
            echo "========================================"

            sh """
            cat > event.json <<EOF
[
  {
    "Source":"pipewatch.jenkins",
    "DetailType":"Pipeline Status",
    "EventBusName":"default",
    "Detail":"{\\"PipelineName\\":\\"PipeWatch Frontend\\",\\"BuildNumber\\":\\"${BUILD_NUMBER}\\",\\"Branch\\":\\"main\\",\\"CurrentStage\\":\\"Failed\\",\\"Status\\":\\"FAILED\\",\\"Environment\\":\\"Production\\",\\"TriggeredBy\\":\\"Jenkins\\",\\"StartTime\\":\\"$(date -u +"%Y-%m-%dT%H:%M:%SZ")\\",\\"Duration\\":0,\\"CommitID\\":\\"${GIT_COMMIT ?: 'Unknown'}\\",\\"JobName\\":\\"${JOB_NAME}\\",\\"BuildURL\\":\\"${BUILD_URL}\\"}"
  }
]
EOF

            aws events put-events --entries file://event.json
            """
        }

        aborted {

            echo "========================================"
            echo "Pipeline Status : ABORTED"
            echo "========================================"

            sh """
            cat > event.json <<EOF
[
  {
    "Source":"pipewatch.jenkins",
    "DetailType":"Pipeline Status",
    "EventBusName":"default",
    "Detail":"{\\"PipelineName\\":\\"PipeWatch Frontend\\",\\"BuildNumber\\":\\"${BUILD_NUMBER}\\",\\"Branch\\":\\"main\\",\\"CurrentStage\\":\\"Aborted\\",\\"Status\\":\\"ABORTED\\",\\"Environment\\":\\"Production\\",\\"TriggeredBy\\":\\"Jenkins\\",\\"StartTime\\":\\"$(date -u +"%Y-%m-%dT%H:%M:%SZ")\\",\\"Duration\\":0,\\"CommitID\\":\\"${GIT_COMMIT ?: 'Unknown'}\\",\\"JobName\\":\\"${JOB_NAME}\\",\\"BuildURL\\":\\"${BUILD_URL}\\"}"
  }
]
EOF

            aws events put-events --entries file://event.json
            """
        }

        always {

            echo "========================================"
            echo "Pipeline Finished"
            echo "========================================"

            cleanWs()
        }
    }
}
