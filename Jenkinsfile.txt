pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Test') {
            steps {
                sh '''
                echo "========================================"
                echo "Running HTML Validation..."
                echo "========================================"

                ERROR=0

                # Check if index.html exists
                if [ ! -f index.html ]; then
                    echo "ERROR: index.html not found"
                    ERROR=1
                fi

                # Validate HTML tags
                grep -q "<html" index.html || {
                    echo "ERROR: Missing <html> tag"
                    ERROR=1
                }

                grep -q "<head" index.html || {
                    echo "ERROR: Missing <head> tag"
                    ERROR=1
                }

                grep -q "<title>" index.html || {
                    echo "ERROR: Missing <title> tag"
                    ERROR=1
                }

                grep -q "<body" index.html || {
                    echo "ERROR: Missing <body> tag"
                    ERROR=1
                }

                if [ $ERROR -eq 1 ]; then
                    echo "========================================"
                    echo "Validation Failed"
                    echo "Pipeline Stopped"
                    echo "========================================"
                    exit 1
                fi

                echo "========================================"
                echo "Validation Successful"
                echo "========================================"
                '''
            }
        }

        stage('Deploy to Amazon S3') {
            steps {
                sh '''
                echo "========================================"
                echo "Deploying index.html to Amazon S3..."
                echo "========================================"

                aws s3 cp index.html s3://s3-deployment-files/index.html

                echo "========================================"
                echo "Deployment Successful"
                echo "========================================"
                '''
            }
        }
    }

    post {

        success {
            echo "========================================"
            echo "Pipeline Status : SUCCESS"
            echo "Website Successfully Deployed to Amazon S3"
            echo "========================================"
        }

        failure {
            echo "========================================"
            echo "Pipeline Status : FAILURE"
            echo "Deployment Skipped"
            echo "========================================"
        }

        always {
            echo "========================================"
            echo "Pipeline Finished"
            echo "========================================"
        }
    }
}
