pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                echo "========== CHECKOUT =========="
                checkout scm
                echo "Source code downloaded successfully."
            }
        }

        stage('Build') {
            steps {
                sh '''
                echo "========== BUILD =========="
                echo "Building Static Website..."
                ls -la
                '''
            }
        }

        stage('Test') {
            steps {
                sh '''
                echo "========== TEST =========="

                # Check main HTML file exists
                if [ ! -f index.html ]; then
                    echo "ERROR: index.html not found"
                    exit 1
                fi

                echo "index.html found"

                # Check HTML structure
                grep -q "<html" index.html || { echo "Missing <html> tag"; exit 1; }
                grep -q "</html>" index.html || { echo "Missing </html> tag"; exit 1; }

                grep -q "<head" index.html || { echo "Missing <head> tag"; exit 1; }
                grep -q "</head>" index.html || { echo "Missing </head> tag"; exit 1; }

                grep -q "<body" index.html || { echo "Missing <body> tag"; exit 1; }
                grep -q "</body>" index.html || { echo "Missing </body> tag"; exit 1; }

                grep -q "<title>" index.html || { echo "Missing <title> tag"; exit 1; }

                echo "HTML Validation Successful"
                '''
            }
        }

        stage('Result') {
            steps {
                echo "Build and Testing Completed Successfully"
            }
        }
    }

    post {

        success {
            echo "Pipeline executed successfully."
        }

        failure {
            echo "Pipeline failed. Please check the errors."
        }

        always {
            echo "Pipeline execution finished."
        }
    }
}
