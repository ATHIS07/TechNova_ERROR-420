pipeline {
    agent any

    environment {
        S3_BUCKET = 's3-deployment-files'
        AWS_DEFAULT_REGION = 'ap-south-1'
    }

    options {
        timestamps()
    }
    
    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Validate HTML') {
            steps {
                sh '''
                echo "========================================"
                echo "Validating HTML..."
                echo "========================================"

                if [ ! -f index.html ]; then
                    echo "ERROR: index.html not found."
                    exit 1
                fi

                tidy -qe index.html

                echo "HTML Validation Passed."
                '''
            }
        }

        stage('Deploy Website') {
            steps {
                sh '''
                echo "========================================"
                echo "Deploying Website to Amazon S3..."
                echo "========================================"

                aws s3 sync . s3://$S3_BUCKET \
                    --delete \
                    --exclude ".git/*" \
                    --exclude ".github/*" \
                    --exclude "*.zip" \
                    --exclude "Jenkinsfile" \
                    --exclude "Jenkinsfile.txt" \
                    --exclude "*.md" \
                    --exclude ".gitignore"

                echo "Deployment Completed Successfully."
                '''
            }
        }

        stage('Publish Event to EventBridge') {
            steps {
                sh """
                echo "========================================"
                echo "Publishing Pipeline Event to EventBridge"
                echo "========================================"

                cat > event.json <<EOF
[
  {
    "Source": "pipewatch.jenkins",
    "DetailType": "Pipeline Status",
    "EventBusName": "default",
    "Detail": "{\\"pipelineName\\":\\"PipeWatch Website\\",\\"buildNumber\\":\\"${BUILD_NUMBER}\\",\\"branch\\":\\"main\\",\\"stage\\":\\"Deploy\\",\\"status\\":\\"SUCCESS\\",\\"environment\\":\\"Production\\",\\"triggeredBy\\":\\"Jenkins\\"}"
  }
]
EOF

                echo "Generated Event:"
                cat event.json

                aws events put-events --entries file://event.json

                echo "Event Published Successfully."
                """
            }
        }

    }

    post {

        success {
            echo "========================================"
            echo "Pipeline Status : SUCCESS"
            echo "Website Successfully Deployed"
            echo "========================================"
        }

        failure {
            echo "========================================"
            echo "Pipeline Status : FAILURE"
            echo "Deployment Cancelled"
            echo "========================================"
        }

        always {
            echo "========================================"
            echo "Pipeline Finished"
            echo "========================================"
            cleanWs()
        }
    }
}
