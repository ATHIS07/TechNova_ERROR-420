pipeline {
    agent any

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Test') {
            steps {
                sh '''
                ERROR=0

                [ -f index.html ] || { echo "ERROR: index.html not found"; ERROR=1; }

                grep -q "<title>" index.html || {
                    echo "ERROR: Missing title"
                    ERROR=1
                }

                if [ $ERROR -eq 1 ]; then
                    exit 1
                fi

                echo "Validation Successful"
                '''
            }
        }

        stage('Deploy') {
            steps {
                sh '''
                sudo cp index.html /var/www/html/
                sudo systemctl restart apache2
                curl http://localhost
                '''
            }
        }

    }
}
