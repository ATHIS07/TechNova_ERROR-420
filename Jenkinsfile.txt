stage('Test') {
    steps {
        sh '''
        echo "========== Running Project Validation =========="

        ERROR=0

        # Check required files
        [ -f index.html ] || { echo "ERROR: index.html not found"; ERROR=1; }
        [ -f style.css ] || { echo "ERROR: style.css not found"; ERROR=1; }
        [ -f script.js ] || { echo "ERROR: script.js not found"; ERROR=1; }

        # Check files are not empty
        [ -s index.html ] || { echo "ERROR: index.html is empty"; ERROR=1; }
        [ -s style.css ] || { echo "ERROR: style.css is empty"; ERROR=1; }
        [ -s script.js ] || { echo "ERROR: script.js is empty"; ERROR=1; }

        # Check HTML title
        grep -q "<title>" index.html || {
            echo "ERROR: Missing <title> tag"
            ERROR=1
        }

        if [ $ERROR -eq 1 ]; then
            echo "Validation Failed"
            exit 1
        else
            echo "Validation Successful"
        fi
        '''
    }
}
